    80	  if (!DASHBOARD_PASSWORD || !token) return false;
    81	  const dot = token.indexOf(".");
    82	  if (dot === -1) return false;
    83	
    84	  const issuedAtStr = token.slice(0, dot);
    85	  const sig = token.slice(dot + 1);
    86	  const issuedAt = parseInt(issuedAtStr, 10);
    87	  if (isNaN(issuedAt)) return false;
    88	
    89	  const age = Math.floor(Date.now() / 1000) - issuedAt;

   181	  const chain = [];

   183	  for (let hop = 0; hop <= maxHops; hop++) {
