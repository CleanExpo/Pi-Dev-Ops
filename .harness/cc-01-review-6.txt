+function riskStyle(risk: string): React.CSSProperties {
+  // Alpha washes of the deck LED fills; text = the AA --deck-*-text variants.

+    fontSize: 12,
+    fontWeight: 600,
+    fontFamily: mono,
+    background: bg,
+    color: fg,
+    border: `1px solid ${bd}`,
+  }
